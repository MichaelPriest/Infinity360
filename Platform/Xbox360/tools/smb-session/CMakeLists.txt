cmake_minimum_required(VERSION 3.16)
project(infinity_smb_session LANGUAGES C CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "Host diagnostic currently supports Linux only")
endif()
set(INFINITY_LIBSMB2_SOURCE "" CACHE PATH "Reviewed libsmb2 checkout")
if(NOT EXISTS "${INFINITY_LIBSMB2_SOURCE}/include/smb2/libsmb2.h")
  message(FATAL_ERROR "Provide pinned libsmb2 checkout 5fdd128012b384ed2a79293611d7f7dbaaa84cc7")
endif()
find_package(Git REQUIRED)
execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${INFINITY_LIBSMB2_SOURCE}" rev-parse HEAD
  OUTPUT_VARIABLE SMB_REV OUTPUT_STRIP_TRAILING_WHITESPACE RESULT_VARIABLE SMB_GIT_RESULT)
if(NOT SMB_GIT_RESULT EQUAL 0 OR NOT SMB_REV STREQUAL "5fdd128012b384ed2a79293611d7f7dbaaa84cc7")
  message(FATAL_ERROR "Unexpected libsmb2 revision")
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(ENABLE_EXAMPLES OFF CACHE BOOL "" FORCE)
set(ENABLE_UTILS OFF CACHE BOOL "" FORCE)
set(ENABLE_LIBDCERPC OFF CACHE BOOL "" FORCE)
set(ENABLE_LIBKRB5 OFF CACHE BOOL "" FORCE)
set(ENABLE_GSSAPI OFF CACHE BOOL "" FORCE)
set(ENABLE_LIBQUIC OFF CACHE BOOL "" FORCE)
set(ENABLE_RDMA OFF CACHE BOOL "" FORCE)
include("../../third_party/libsmb2-xsdk/Port.cmake")
set(SMB_PREPARED "${CMAKE_CURRENT_BINARY_DIR}/infinity-libsmb2-host-src")
infinity_prepare_smb2("${INFINITY_LIBSMB2_SOURCE}" "${SMB_PREPARED}" host)
add_subdirectory("${SMB_PREPARED}" libsmb2 EXCLUDE_FROM_ALL)
add_executable(infinity_smb_session main.cpp)
target_compile_features(infinity_smb_session PRIVATE cxx_std_17)
target_link_libraries(infinity_smb_session PRIVATE smb2)
enable_testing()
add_test(NAME smb_session_validation COMMAND infinity_smb_session --self-test)

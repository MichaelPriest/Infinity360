# Cloud360 Xbox 360 build via xenosworks/xsdk (independent from AuroraX XDK project).
cmake_minimum_required(VERSION 3.25)
project(cloud360 LANGUAGES C)
option(CLOUD360_WITH_LOGOS "Compile the optional offline logo renderer; requires glslang-xenon" OFF)
if(CLOUD360_WITH_LOGOS)
    add_executable(cloud360 main.c logo.c ../core/cloud360.c)
else()
    add_executable(cloud360 main.c logo_stub.c ../core/cloud360.c)
endif()
set_target_properties(cloud360 PROPERTIES C_STANDARD 17 C_EXTENSIONS ON)
target_include_directories(cloud360 PRIVATE ../core)
target_compile_options(cloud360 PRIVATE -Wall -Wextra)
if(CLOUD360_WITH_LOGOS)
    xsdk_add_shaders(cloud360 logo.vert logo.frag)
endif()
xsdk_add_xex(cloud360 TITLE_ID 0x43333630 NAME "Cloud360 IPTV" VERSION 0.5.0.0)

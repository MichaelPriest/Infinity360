cmake_minimum_required(VERSION 3.16)
project(Cloud360Catalog LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_library(cloud360_core core/cloud360.c core/codec_probe.c)
target_include_directories(cloud360_core PUBLIC core)
if(MSVC)
  target_compile_options(cloud360_core PRIVATE /W4)
else()
  target_compile_options(cloud360_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
enable_testing()
add_executable(cloud360_test tests/test_catalog.c)
target_link_libraries(cloud360_test PRIVATE cloud360_core)
target_compile_definitions(cloud360_test PRIVATE CLOUD360_FIXTURE="fixtures/demo.m3u")
add_test(NAME cloud360_catalog COMMAND cloud360_test)
set_tests_properties(cloud360_catalog PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(cloud360_codec_probe_test tests/test_codec_probe.c)
target_link_libraries(cloud360_codec_probe_test PRIVATE cloud360_core)
add_test(NAME cloud360_codec_probe COMMAND cloud360_codec_probe_test)

option(CLOUD360_WITH_HOST_FFMPEG "Compile real FFmpeg software decoder on host" OFF)
if(CLOUD360_WITH_HOST_FFMPEG)
  enable_language(CXX)
  set(CMAKE_CXX_STANDARD 17)
  set(CMAKE_CXX_STANDARD_REQUIRED ON)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(AVCODEC REQUIRED IMPORTED_TARGET libavformat libavcodec libavutil)
  add_library(cloud360_ffmpeg_decoder core/ffmpeg_decoder.c core/offline_decode.c core/container_decode.c core/frame_output.c)
  target_include_directories(cloud360_ffmpeg_decoder PUBLIC core)
  target_link_libraries(cloud360_ffmpeg_decoder PUBLIC PkgConfig::AVCODEC cloud360_core)
  add_executable(cloud360_decoder_init_test tests/test_ffmpeg_init.c)
  target_link_libraries(cloud360_decoder_init_test PRIVATE cloud360_ffmpeg_decoder)
  add_test(NAME cloud360_ffmpeg_init COMMAND cloud360_decoder_init_test)
  add_executable(cloud360_offline_decode_test tests/test_offline_decode.c)
  target_link_libraries(cloud360_offline_decode_test PRIVATE cloud360_ffmpeg_decoder)
  add_executable(cloud360_container_test tests/test_container_decode.c)
  target_link_libraries(cloud360_container_test PRIVATE cloud360_ffmpeg_decoder)
  foreach(container mp4 ts)
    add_test(NAME cloud360_container_${container} COMMAND cloud360_container_test "${CMAKE_CURRENT_SOURCE_DIR}/fixtures/generated/test.${container}")
  endforeach()
  add_executable(cloud360_av_timeline_test tests/test_av_timeline.cpp)
  target_link_libraries(cloud360_av_timeline_test PRIVATE cloud360_ffmpeg_decoder)
  foreach(container mp4 ts)
    add_test(NAME cloud360_av_timeline_${container} COMMAND cloud360_av_timeline_test "${CMAKE_CURRENT_SOURCE_DIR}/fixtures/generated/test.${container}")
  endforeach()
  add_executable(cloud360_incremental_test tests/test_container_incremental.cpp)
  target_link_libraries(cloud360_incremental_test PRIVATE cloud360_ffmpeg_decoder)
  foreach(container mp4 ts)
    add_test(NAME cloud360_incremental_${container} COMMAND cloud360_incremental_test "${CMAKE_CURRENT_SOURCE_DIR}/fixtures/generated/long.${container}")
  endforeach()
  add_executable(cloud360_stream_queue_test tests/test_av_stream_queue.cpp)
  target_link_libraries(cloud360_stream_queue_test PRIVATE cloud360_ffmpeg_decoder)
  foreach(container mp4 ts)
    add_test(NAME cloud360_stream_queue_${container} COMMAND cloud360_stream_queue_test "${CMAKE_CURRENT_SOURCE_DIR}/fixtures/generated/long.${container}")
  endforeach()
  add_executable(cloud360_reader_test tests/test_container_reader.cpp)
  target_link_libraries(cloud360_reader_test PRIVATE cloud360_ffmpeg_decoder)
  add_test(NAME cloud360_reader_ts COMMAND cloud360_reader_test "${CMAKE_CURRENT_SOURCE_DIR}/fixtures/generated/long.ts")
  add_test(NAME cloud360_container_reject_playlist COMMAND cloud360_container_test "${CMAKE_CURRENT_SOURCE_DIR}/fixtures/demo.m3u")
  set_tests_properties(cloud360_container_reject_playlist PROPERTIES WILL_FAIL TRUE)
  foreach(codec h264 aac mp3 mpeg2 ac3)
    add_test(NAME cloud360_decode_${codec}
      COMMAND cloud360_offline_decode_test "${CMAKE_CURRENT_SOURCE_DIR}/fixtures/generated/test.${codec}" "${codec}")
  endforeach()
endif()

cmake_minimum_required(VERSION 3.16)
project(Cloud360Catalog LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_library(cloud360_core core/cloud360.c)
target_include_directories(cloud360_core PUBLIC core)
if(MSVC)
  target_compile_options(cloud360_core PRIVATE /W4)
else()
  target_compile_options(cloud360_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
enable_testing()
add_executable(cloud360_test tests/test_catalog.c)
target_link_libraries(cloud360_test PRIVATE cloud360_core)
target_compile_definitions(cloud360_test PRIVATE CLOUD360_FIXTURE="fixtures/demo.m3u")
add_test(NAME cloud360_catalog COMMAND cloud360_test)
set_tests_properties(cloud360_catalog PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
